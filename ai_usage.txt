Крок 3a: додаємо клас CodeGen (старий код ще не чіпаємо)

bash
cat > /tmp/codegen_snippet.py << 'EOF'
class CodeGen:
    def __init__(self, builder):
        self.builder = builder
        self.module = builder.module
        self.function = builder.function

    def generate(self, tree):
        tree.accept(self)

    def visit_program(self, node):
        for stmt in node.statements:
            stmt.accept(self)
        node.exit_node.accept(self)

    def visit_decl(self, node):
        llvm_ty = LLVM_TYPES[node.type_name]
        value = node.init.accept(self)
        value = coerce(self.builder, value, node.init.type, node.type_name)
        node.ptr = self.builder.alloca(llvm_ty, name=node.name)
        self.builder.store(value, node.ptr)

    def visit_assign(self, node):
        value = node.value.accept(self)
        value = coerce(self.builder, value, node.value.type, node.decl.type_name)
        self.builder.store(value, node.decl.ptr)

    def visit_exit(self, node):
        value = node.value.accept(self)
        printf = self.module.get_global("printf")

        if node.value.type == "bool":
            fmt = self.module.get_global("fmt_bool")
            true_ptr = self.builder.bitcast(self.module.get_global("true_str"), ir.PointerType(I8))
            false_ptr = self.builder.bitcast(self.module.get_global("false_str"), ir.PointerType(I8))
            chosen = self.builder.select(value, true_ptr, false_ptr)
            self.builder.call(printf, [self.builder.bitcast(fmt, ir.PointerType(I8)), chosen])
        else:
            wide = coerce(self.builder, value, node.value.type, "i64")
            fmt = self.module.get_global("fmt")
            self.builder.call(printf, [self.builder.bitcast(fmt, ir.PointerType(I8)), wide])

        self.builder.ret(ir.Constant(I32, 0))

    def visit_binop(self, node):
        lhs = node.left.accept(self)
        rhs = node.right.accept(self)

        if node.op in ("+", "-", "*"):
            lhs = coerce(self.builder, lhs, node.left.type, node.type)
            rhs = coerce(self.builder, rhs, node.right.type, node.type)
            if node.op == "+":
                return self.builder.add(lhs, rhs)
            elif node.op == "-":
                return self.builder.sub(lhs, rhs)
            else:
                return self.builder.mul(lhs, rhs)
        else:
            common = "i64" if "i64" in (node.left.type, node.right.type) else node.left.type
            lhs = coerce(self.builder, lhs, node.left.type, common)
            rhs = coerce(self.builder, rhs, node.right.type, common)
            pred = "==" if node.op == "==" else "!="
            return self.builder.icmp_signed(pred, lhs, rhs)

    def visit_var(self, node):
        return self.builder.load(node.decl.ptr)

    def visit_const(self, node):
        return ir.Constant(LLVM_TYPES[node.type], int(node.value))

    def visit_bool(self, node):
        return ir.Constant(I1, int(node.value))

Крок 3c: прибираємо мертві codegen() з нод

Це 8 однотипних видалень, тому робимо одноразовим скриптом у /tmp (не в репозиторії). Він знаходить методи codegen лише в класах *Node і вирізає їх.

bash
cat > /tmp/strip.py << 'EOF'
import ast
src = open("compiler.py").read()
lines = src.split("\n")
tree = ast.parse(src)
spans = []
for cls in tree.body:
    if isinstance(cls, ast.ClassDef) and cls.name.endswith("Node"):
        for item in cls.body:
            if isinstance(item, ast.FunctionDef) and item.name == "codegen":
                spans.append((item.lineno, item.end_lineno))
for start, end in sorted(spans, reverse=True):
    del lines[start - 1:end]
    if start - 1 < len(lines) and lines[start - 1] == "":
        del lines[start - 1]
open("compiler.py", "w").write("\n".join(lines))
print(f"removed {len(spans)} codegen methods")
EOF

bool b{false}
bool c{!a}
bool mut d{! b}
d := !c
if !c
{
exit d
}
exit 0
Expected: Program exit with result true.
Lexer
if and else are keywords. After ! the machine looks at one byte: = makes !=, anything else makes ! and is read
again — the same move as the byte that ends an identifier. And the Practice 2 rule that { must close on its line
leaves the lexer: { and } are plain tokens now, and the parser pairs them. i32 x{5 is still an error — the parser’s,
at the same position.
Grammar and tree
program ::= { statement } exit
statement ::= decl | assign | if
if ::= "if" expr NL block [ "else" NL block ]
block ::= "{" NL { statement } [ exit NL ] "}" NL (not empty)
factor ::= number | "true" | "false" | ident | "!" factor
NL is the end of a line. A statement was a line; an if is several. Give the parser a line cursor next to the token
cursor: peek_line looks at the next line, next_line makes it the current one. parse_statement consumes one
line — or, for if, the condition line, the { line, every line of the block and the } line. Two new nodes: If (condition,
then-block, optional else-block) and Block (statements, optional exit); one new expression node, Not. The dump
of the nested program above:

Крок 1.4: курсор по рядках у парсері
Зараз parse_program ходить по self.lines циклом for, і кожен statement це рівно один рядок. if займає кілька рядків, тому парсеру потрібен курсор по рядках, який можна пересувати з будь-якого методу (з parse_if, parse_block). Цей крок нічого не змінює в поведінці: це рефакторинг, який має лишити всі 61 тест зеленими. Я перевірила в себе: старий і новий парсер видають однаковий результат на 20+ програмах у трьох режимах.

A. Parser.__init__. Було:

python
    def __init__(self, lines):
        self.lines, self.toks, self.pos = lines, [], 0
стало:

python
    def __init__(self, lines):
        self.lines, self.toks, self.pos = lines, [], 0
        self.line_idx = 0

    def peek_line(self):
        while self.line_idx < len(self.lines) and not self.lines[self.line_idx]:
            self.line_idx += 1
        if self.line_idx < len(self.lines):
            return self.lines[self.line_idx]
        return None

    def next_line(self):
        toks = self.peek_line()
        if toks is None:
            return None
        self.line_idx += 1
        self.toks, self.pos = toks, 0
        return toks

    def expect_eol(self):
        tok = self.peek()
        if tok is not None:
            self.error(f"unexpected '{tok.text}' after the statement")
Що тут нового:

line_idx це індекс наступного непрочитаного рядка. Токенний курсор (toks, pos) лишається для поточного рядка.
peek_line() дивиться на наступний непорожній рядок, не з'їдаючи його. Порожні рядки пропускаються: лексер дає для них порожні списки, а мова дозволяє порожні рядки будь-де.
next_line() робить цей рядок поточним (toks, pos = 0) і повертає його. Якщо рядків більше нема, повертає None, а toks лишається попереднім, повністю прочитаним рядком. Це важливо для позиції помилки «знайдено кінець рядка».

## Крок 2.2: `visit_block`, `visit_if`, `visit_not`

Додай три методи в `SemanticChecker`, наприклад після `visit_bool`:

```python
    def visit_block(self, node):
        self.scopes.append({})          # новий кадр на вході в блок
        for stmt in node.statements:
            stmt.accept(self)
        if node.exit_node is not None:
            node.exit_node.accept(self)
        self.scopes.pop()               # кадр зникає на виході

    def visit_if(self, node):
        cond_type = node.condition.accept(self)
        if cond_type != "bool":
            raise CompileError(
                f"line {node.line}:{node.col}: the condition of 'if' must be bool, got {cond_type}")
        node.then_block.accept(self)
        if node.else_block is not None:
            node.else_block.accept(self)

    def visit_not(self, node):
        operand_type = node.operand.accept(self)
        if operand_type != "bool":
            raise CompileError(
                f"line {node.line}:{node.col}: cannot apply '!' to {operand_type}")
        node.type = "bool"
        return node.type
```

- Кадр створюється для кожного блоку окремо. Тому `then` і `else` не бачать змінних одне одного, а змінна з блоку не видна після `}`.
- `ProgramNode` лишається на глобальному кадрі `scopes[0]`.
- Умова `if` має бути саме `bool`. `if x` з `i32` має давати помилку.

## Навіщо цей крок

У кроці 2.1 ми замінили одну плоску таблицю змінних на **стек кадрів** (`self.scopes`). Але поки що ніхто не додає в нього нові кадри. Кадр `scopes[0]` глобальний, і єдиний, що існує. Крок 2.2 вчить чекер **відкривати кадр, коли заходить у `{ ... }`, і закривати, коли виходить**.

## Як працює стек

```
i32 mut x = 1          # scopes = [ {x} ]
if cond
{                      # visit_block: append({})  ->  [ {x}, {} ]
    i32 y = 2          #                              [ {x}, {y} ]
    x = y              # lookup('x') іде зверху вниз: {y} -> {x}, знайшов
}                      # pop()                    ->  [ {x} ]
y = 3                  # lookup('y'): у {x} немає -> помилка "used before its declaration"
```

Тобто змінна, оголошена в блоці, зникає після `}`. Внутрішній блок бачить зовнішні змінні, а зовнішній внутрішніх не бачить.

## Що робить кожен метод

**`visit_block`**
- `scopes.append({})` створює порожній кадр для цього блоку.
- Потім обходить `statements`, а в кінці `exit_node`, якщо він є. Усе це відбувається всередині нового кадру.
- `scopes.pop()` видаляє кадр. Після цього всі змінні блоку забуті.
- Кожен блок має свій власний кадр. Тому `then` і `else` не бачать змінних одне одного: після `then` кадр знято, а для `else` створено новий.

**`visit_if`**
- `node.condition.accept(self)` повертає тип умови. Це працює, бо `visit_var`, `visit_binop` і `visit_bool` вже повертають тип.
- Якщо тип не `bool`, кидаємо помилку з позицією `if`. Це та помилка зі завдання: `if x` з `i32` не можна.
- Далі перевіряємо `then_block`, а `else_block` лише якщо він існує (він необов'язковий).

**`visit_not`**
- Перевіряє, що операнд `!` має тип `bool`. Для `!5` кидає помилку "cannot apply '!' to i32".
- Записує `node.type = "bool"` і повертає його. Результат `!` завжди `bool`, і так він може стояти в умові `if` або у виразі.

## Чому це правильно саме так

- **Shadowing працює «безкоштовно».** У внутрішньому блоці можна оголосити `x`, навіть якщо `x` є зовні. Дубль перевіряється лише в `scopes[-1]`, тобто тільки в поточному блоці. Це вже зроблено в 2.1.
- **Помилка про використання до оголошення** теж працює. Якщо змінну оголосили в блоці, а використали після `}`, `lookup` не знайде її: кадр уже знятий.
- **Пам'ять про «кому належить змінна»** лишається в `node.decl`. Пізніше CodeGen бере `node.decl.ptr`, тому вкладені блоки не потребують додаткового коду для імен.

Якщо хочеш, можу пояснити окремо, чому `pop()` потрібен навіть у разі помилки, або показати приклад із shadowing. Або вставляй методи й пиши "готово", тоді підемо на 2.3.

## Що ми зробили

Ми поміняли місцями два рядки в `main_cli`.

**Було:**
```python
tree = Parser(token_lines).parse_program()
if mode == "--ast":
    tree.dump()        # друкуємо дерево...
    return
SemanticChecker().check(tree)   # ...а перевірка тільки після
```

**Стало:**
```python
tree = Parser(token_lines).parse_program()
SemanticChecker().check(tree)   # спочатку семантика
if mode == "--ast":
    tree.dump()
    return
```

## Чому це потрібно

**1. `--ast` має показувати лише коректні програми.**
Раніше `--ast` зупинявся одразу після парсера, тому семантичні помилки не ловилися. У завданні сказано: "Show: `--ast` for the warm-up program" (його треба показати і прийняти). А п'ять програм з помилками мають бути відхилені з `line:column`. Тепер усі режими поводяться однаково: `--ast` і звичайна компіляція відхиляють однакові програми.

**2. Чекер ставить `node.decl` і `node.type`.**
`dump()` друкує дерево вже після того, як чекер пройшов по ньому. Метод `visit_not` записує `node.type`, а `visit_var` і `visit_assign` ставлять `node.decl`. Для поточних міток дампа (`Var x`, `Const 10`) це нічого не змінює, але порядок правильний: дерево показується в тому стані, в якому його побачить кодогенератор.

**3. Помилки йдуть через ту саму обробку.**
`SemanticChecker` кидає `CompileError`. Він ловиться в тому самому `try/except`, який друкує `compilation error: ...` у stderr і виходить з кодом 1. Нічого додатково писати не треба.

## Що показала перевірка

`py_compile` пройшов, і `run_tests.py` дав `61 passed, 0 failed` без жодного `FAIL`. Тобто жоден існуючий `.ast`-тест не покладався на дамп до чекера. Нічого не зламалося.

Раніше, при переносі `--ast` перед чекером, ми бачили зайві `DIFF` у diffcheck, бо `--ast` для програм з семантичними помилками раніше проходив. Тепер це повертається назад: `--ast` на такій програмі видасть помилку, і це правильно.

## Крок 3.1: хелпер `new_slot`

У класі `CodeGen` заміни `__init__` і додай метод одразу після нього:

```python
class CodeGen:
    def __init__(self, builder):
        self.builder = builder
        self.module = builder.module
        self.function = builder.function
        self.n_slots = 0

    def new_slot(self, llvm_ty, name):
        entry = self.function.entry_basic_block
        saved = self.builder.block
        if self.n_slots < len(entry.instructions):
            self.builder.position_before(entry.instructions[self.n_slots])
        else:
            self.builder.position_at_end(entry)
        ptr = self.builder.alloca(llvm_ty, name=name)
        self.n_slots += 1
        self.builder.position_at_end(saved)
        return ptr
```

Як це працює: `n_slots` рахує вже створені `alloca`. Вони лежать на початку `entry`, тому новий слот вставляється перед першою не-`alloca` інструкцією, або в кінець `entry`, якщо інших інструкцій ще немає. Потім білдер повертається в той блок, де був (`saved`). Завдяки цьому порядок слотів відповідає порядку оголошень.

У `visit_decl` заміни один рядок:

```python
        node.ptr = self.new_slot(llvm_ty, node.name)
```

(було `self.builder.alloca(llvm_ty, name=node.name)`).

## Крок 3.2: `visit_not`, `visit_block`, `visit_if`

У `CodeGen` після `visit_bool` додай:

```python
    def visit_not(self, node):
        return self.builder.not_(node.operand.accept(self))

    def visit_block(self, node):
        for stmt in node.statements:
            stmt.accept(self)
        if node.exit_node is not None:
            node.exit_node.accept(self)

    def visit_if(self, node):
        cond = node.condition.accept(self)
        then_bb = self.function.append_basic_block("then")
        else_bb = self.function.append_basic_block("else") if node.else_block else None
        merge_bb = self.function.append_basic_block("merge")
        self.builder.cbranch(cond, then_bb, else_bb or merge_bb)

        self.builder.position_at_end(then_bb)
        node.then_block.accept(self)
        if not self.builder.block.is_terminated:
            self.builder.branch(merge_bb)

        if else_bb is not None:
            self.builder.position_at_end(else_bb)
            node.else_block.accept(self)
            if not self.builder.block.is_terminated:
                self.builder.branch(merge_bb)

        self.builder.position_at_end(merge_bb)
```

Що тут важливо:
- Перевірка `is_terminated` читає **поточний** блок (`self.builder.block`), а не `then_bb`. Вкладений `if` пересунув білдер у свій `merge`, і `br` треба дописати саме туди.
- Якщо арка виконала `exit`, у ній уже є `ret`, і `br` не додається. Інакше було б два термінатори.
- Без `else` `cbranch` веде у `merge`: умова хибна, і `then` пропускається.
- `not_` для `i1` це `xor` з `true`.

## Спершу: що таке basic block і builder

Функція `main` в IR це не один рядок інструкцій, а **набір basic block'ів**. Кожен блок має мітку, список інструкцій і закінчується одним термінатором: `br` (перехід), `cbranch` (умовний перехід) або `ret`. Всередині блоку немає переходів, тому він завжди виконується від початку до кінця.

`self.builder` — це "курсор": куди він зараз поставлений, туди й потрапляє наступна інструкція. До цього практика мала один блок `entry`, і курсор ніколи не рухався. `if` вперше змушує його рухатись.

## Крок 3.1: чому `new_slot`

`alloca` резервує пам'ять під змінну. Нам потрібно, щоб усі `alloca` лежали в `entry`, з двох причин, які названо в завданні:

1. **Шляхи виконання.** Якщо `alloca` лежить усередині `then`, то коли виконується `else`, цього слота не існує. А `exit x` після `if` читав би неіснуючу пам'ять.
2. **`mem2reg`.** LLVM перетворює змінні на регістри (і будує `phi`) тільки для `alloca` в `entry`.

Але `visit_decl` викликається тоді, коли білдер стоїть десь у `then`, `else` чи `merge`. Тому `new_slot` робить три дії:

```
запам'ятати, де стоїть білдер   (saved)
перейти в entry і створити alloca
повернутись у saved
```

Деталь із `n_slots`: вставка в кінець `entry` не працює. Туди вже може бути дописаний `cbranch`, а нова інструкція після термінатора недопустима. Тому ми вставляємо **перед** першою не-`alloca` інструкцією (`position_before(...)`). `n_slots` рахує, скільки `alloca` вже стоїть зверху, щоб знати, де закінчується їхня ділянка.

Тому в тесті на кроці 3.1 нічого не змінилось: у старих програмах немає `if`, і єдина різниця в тому, що всі `alloca` стоять зверху `entry`.

## Крок 3.2: що робить `visit_if`

Приклад із прототипу, warm-up (скорочено):

```
entry:   alloca x, x.1, x.2 · store 10 · br i1 true → then / merge
then:    store true → x.1 · load · br i1 %cond → then.1 / merge.1
merge:   load x · printf · ret            ← останній exit програми
then.1:  store 20 → x.2 · printf · ret    ← exit усередині блоку
merge.1: load x.1 · printf · ret          ← exit після внутрішнього if
```

Кроки `visit_if`:

1. **Обчислити умову** в поточному блоці. Це `i1`.
2. **Створити блоки** `then`, `else` (якщо є), `merge`. Вони поки порожні і просто додані у функцію.
3. **`cbranch(cond, then_bb, else_bb or merge_bb)`**. Він закриває поточний блок: далі писати в нього не можна. Без `else` умова хибна → одразу `merge`. Зверни увагу: `cbranch` приймає **об'єкти блоків**, а не текстові мітки (за текстові мітки завдання ставить 0).
4. **Пересунути білдер у `then`** (`position_at_end`) і згенерувати тіло блоку.
5. **Закрити `then`.** Якщо блок ще не завершений, дописати `br merge`. Якщо всередині був `exit`, там уже є `ret`, і другого термінатора додавати не можна.
6. **Те саме для `else`.**
7. **Поставити білдер у `merge`.** Усе, що йде після `if`, потрапляє туди.

## Чому перевіряємо `self.builder.block`, а не `then_bb`

Припустимо, у `then` є вкладений `if`. Після нього білдер стоїть **не в `then`, а в `merge.1`** (merge внутрішнього `if`). Якщо перевіряти `then_bb.is_terminated`, ми побачили б там `cbranch` і не дописали б `br` у `merge.1`. Блок лишився б без термінатора, а це невалідний IR. `self.builder.block` завжди вказує на те місце, де білдер зараз.

## Чому `merge` залишається валідним, коли обидві гілки роблять `exit`

Тоді в `merge` ніхто не входить, але він усе одно отримує останній `exit` програми з `ret`. Блок без передників у LLVM допустимий, і верифікатор його приймає. Так у warm-up `merge` стоїть у списку, хоч до нього й не доходить.

## Чому `visit_not` це одна рядок

`not_` на `i1` генерує `xor i1 %v, true`. Операнд уже обчислений через `accept`, і в нього тип `i1`.

Якщо щось із цього лишається неясним, скажи, що саме (`new_slot`, порядок блоків чи термінатори). Або вставляй код і запускай перевірку з попереднього кроку.

## Крок 4.2: вузол `WhileNode`

Вставляй перед `class ExitNode(Node):`:

```python
class WhileNode(StmtNode):
    def __init__(self, line, col, condition, body):
        super().__init__(line, col)
        self.condition = condition
        self.body = body

    def label(self):
        return "While"

    def children(self):
        return [self.condition, self.body]

    def accept(self, visitor):
        return visitor.visit_while(self)

```

## Крок 4.3: парсер

Перед `def parse_statement(self):` додай метод:

```python
    def parse_while(self):
        while_tok = self.eat()
        condition = self.parse_expr()
        self.expect_eol()
        body = self.parse_block("'while'")
        return WhileNode(while_tok.line, while_tok.col, condition, body)

```

У `parse_statement` додай гілку **перед** перевіркою `else`, тобто одразу після гілки з `if`:

```python
        if tok.kind == "keyword" and tok.text == "while":
            return self.parse_while()
```

## Крок 4.4: чекер

У `SemanticChecker` перед `visit_not` додай:

```python
    def visit_while(self, node):
        cond_type = node.condition.accept(self)
        if cond_type != "bool":
            raise CompileError(
                f"line {node.line}:{node.col}: the condition of 'while' must be bool, got {cond_type}")
        node.body.accept(self)

```

Кадр для тіла створює вже `visit_block`, окремого коду для scope не треба. Це і є "the same scope frame" із завдання.

## Крок 4.5: кодогенерація

У `CodeGen` після `visit_if` (перед порожніми рядками, що ведуть до `def lex`) додай:

```python
    def visit_while(self, node):
        cond_bb = self.function.append_basic_block("while.cond")
        body_bb = self.function.append_basic_block("while.body")
        end_bb = self.function.append_basic_block("while.end")
        self.builder.branch(cond_bb)

        self.builder.position_at_end(cond_bb)
        cond = node.condition.accept(self)
        self.builder.cbranch(cond, body_bb, end_bb)

        self.builder.position_at_end(body_bb)
        node.body.accept(self)
        if not self.builder.block.is_terminated:
            self.builder.branch(cond_bb)

        self.builder.position_at_end(end_bb)
```

Що тут важливо:
- Умова обчислюється **всередині `while.cond`**, а не до переходу. Інакше цикл не перечитував би змінні на кожній ітерації.
- Перед `while.cond` потрібен `br`, бо поточний блок (наприклад `entry`) мусить закінчитись термінатором.
- Кінець тіла повертається на `while.cond` (зворотне ребро), якщо тіло не завершилось `exit`. Та сама перевірка `is_terminated`, що в `if`.
- Слоти (`new_slot`) не змінюються: `alloca` усе ще в `entry`.

# Languages and Compilers Design — Deneshchuk

## Current status
A compiler for a small language with `i32`/`i64`/`bool` variable
declarations (const by default, `mut` for mutable), assignments, `exit`,
and comparisons. Source is tokenized by a hand-written lexer, parsed by a
hand-written recursive-descent parser into an AST, checked by a
`SemanticChecker` visitor that resolves names and types before any LLVM
instruction exists, and compiled to LLVM IR via llvmlite by a walk over
that tree. The initialiser and the right side of `:=` accept a chain of
`+`, `-`, `*` with real precedence and left associativity, optionally
followed by one `==`/`!=` comparison. An `i32` value widens into `i64`
where needed (`sext`); nothing else converts.

## Run the compiler
    python3 compiler.py input.txt output.ll      # compile
    python3 compiler.py --ast input.txt           # print the AST, write nothing
    python3 compiler.py --tokens input.txt         # print the token list, write nothing
    lli output.ll                                  # quick run, no linking
    llc -filetype=obj -relocation-model=pic output.ll -o output.o
    clang -fPIE output.o -o program && ./program

## Language
    i32 x{5}                # const declaration, initialiser mandatory
    i64 mut y{10}            # mutable declaration
    bool b{true}             # bool declaration, true/false only
    y := x + 3                # assignment (mut only), same rules as an initialiser
    i32 z{2 + 3 * 4}          # chains: * binds tighter than + and -
    bool r{x == y}            # one comparison per expression, weaker than + - *
    exit y                    # exit y, exit 42, or exit b; must be the last line

## Type rules
A decimal constant takes the narrowest type it fits (`i32`, else `i64`,
else an error). `+ - *` take two integers and produce the wider type; a
`bool` operand is an error. `==`/`!=` take two integers of any width or
two `bool`s and produce `bool`; mixing `bool` with an integer is an
error. An `i32` may initialise or be assigned to an `i64` variable;
nothing else narrows or crosses the integer/`bool` line.

## Run tests
    python3 run_tests.py

Runs every `tests/*.txt`, `tests/ok/*.txt`, and `tests/err/*.txt` against
its `.expected`: compiles, runs the result through `lli` when compilation
succeeds, and otherwise compares the compiler's stderr. Prints a
PASS/FAIL line per test and a summary; exits non-zero if anything failed.

Valid programs write output.ll and produce no stderr. Invalid programs
print `compilation error: line L:C: ...` to stderr and exit non-zero,
writing no output file.

```markdown
# Languages and Compilers Design — Deneshchuk

## Current status
A compiler for a small language with `i32`/`i64`/`bool` variable
declarations (const by default, `mut` for mutable), assignments, `exit`,
comparisons, `!`, `if`/`else` and `while` with blocks. Source is
tokenized by a hand-written lexer, parsed by a hand-written
recursive-descent parser into an AST, checked by a `SemanticChecker`
visitor that resolves names, scopes and types before any LLVM
instruction exists, and compiled to LLVM IR via llvmlite by a walk over
that tree. The initialiser and the right side of `:=` accept a chain of
`+`, `-`, `*` with real precedence and left associativity, optionally
followed by one `==`/`!=` comparison. An `i32` value widens into `i64`
where needed (`sext`); nothing else converts.

## Run the compiler
    python3 compiler.py input.txt output.ll      # compile
    python3 compiler.py --ast input.txt           # print the AST, write nothing
    python3 compiler.py --tokens input.txt         # print the token list, write nothing
    lli output.ll                                  # quick run, no linking
    opt -passes=mem2reg -S output.ll               # same IR, allocas promoted to registers (phi)
    llc -filetype=obj -relocation-model=pic output.ll -o output.o
    clang -fPIE output.o -o program && ./program

`--ast` runs the semantic pass first, so a program that does not type-check
is rejected in that mode too.

## Language
    i32 x{5}                # const declaration, initialiser mandatory
    i64 mut y{10}            # mutable declaration
    bool b{true}             # bool declaration, true/false only
    y := x + 3                # assignment (mut only), same rules as an initialiser
    i32 z{2 + 3 * 4}          # chains: * binds tighter than + and -
    bool r{x == y}            # one comparison per expression, weaker than + - *
    bool n{!b}                # ! negates a bool; applies to the factor right after it
    exit y                    # exit y, exit 42, exit b or exit !b

Control flow (a block is `{`, one or more lines, `}`, each on its own line):

    if b                      # the condition is on the line of the if
    {
        y := y + 1
    }
    else                      # optional
    {
        y := y - 1
    }

    while i != 11             # same layout; the body returns to the condition
    {
        sum := sum + i
        i := i + 1
    }

A block holds at least one line and may end with `exit` as its last line;
nothing follows an `exit` inside the same block. Blocks nest, and an `if`
or `while` is a statement, so it may stand inside another block. A program
is a list of statements ending with an `exit`; nothing follows it. `{` and
`}` are plain tokens: the lexer does not pair them, the parser does.

## Type rules
A decimal constant takes the narrowest type it fits (`i32`, else `i64`,
else an error). `+ - *` take two integers and produce the wider type; a
`bool` operand is an error. `==`/`!=` take two integers of any width or
two `bool`s and produce `bool`; mixing `bool` with an integer is an
error. `!` takes a `bool`. The condition of `if` and `while` must be
`bool`. An `i32` may initialise or be assigned to an `i64` variable;
nothing else narrows or crosses the integer/`bool` line.

## Scopes
Every block is a scope of its own. The semantic pass keeps a stack of
frames: a block pushes an empty frame on entry and pops it on exit. A
declaration goes into the top frame, and only the top frame is checked
for a duplicate, so an inner block may declare a name again with any type
(shadowing). A use is looked up from the top frame down and takes the
first hit; a name that is in no frame (including one whose block has
ended) is "used before its declaration". Every `Var` and `Assign` node
gets `node.decl`, the declaration it resolved to, so the code generator
never looks a name up.

## Code generation
Every `if` becomes `then`, `else` (when there is one) and `merge` basic
blocks and one conditional branch on the `i1` the condition produced; a
`while` becomes `while.cond`, `while.body` and `while.end` with a branch
back to the condition. Each block ends with exactly one terminator. A
block that ends in `exit` has its `ret` and gets no extra `br`. Every
`alloca` goes into the entry block, whichever block the declaration is
in, so `opt -passes=mem2reg` can promote the slots to registers and build
the `phi` for a variable assigned in both arms of an `if`.

## Run tests
    python3 run_tests.py

Runs every `tests/*.txt`, `tests/ok/*.txt`, and `tests/err/*.txt` against
its `.expected`: compiles, runs the result through `lli` when compilation
succeeds, and otherwise compares the compiler's stderr. Every `.ast`
snapshot is compared with the output of `--ast` for the program next to
it. Prints a PASS/FAIL line per test and a summary; exits non-zero if
anything failed.

Valid programs write output.ll and produce no stderr. Invalid programs
print `compilation error: line L:C: ...` to stderr and exit non-zero,
writing no output file.
```
